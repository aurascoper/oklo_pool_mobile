== digitization script (run with oklo_heartbeat/.venv python -I; PIL and numpy from that venv)
"""Digitize bar endpoints from GW Project Figure 32 (after Freeze and Cherry 1979)
on its k (cm^2) axis. Reproducible: prints the axis calibration and the pixel
endpoints of each bar, then the endpoints in m^2 (1 cm^2 = 1e-4 m^2)."""
import sys
import numpy as np
from PIL import Image

img = np.asarray(Image.open(sys.argv[1]).convert("RGB")).astype(int)
H, W, _ = img.shape
print("image", W, "x", H)
r, g, b = img[..., 0], img[..., 1], img[..., 2]
dark = (r < 90) & (g < 90) & (b < 90)
blue = (b > 150) & (r < 90) & (g < 120)
ink = dark | blue

# Axis lines: rows in the lower part where most columns in the plot span are dark.
span = slice(int(W * 0.17), int(W * 0.97))
frac = dark[:, span].mean(axis=1)
axis_rows = [y for y in range(int(H * 0.6), H) if frac[y] > 0.8]
groups = []
for y in axis_rows:
    if groups and y - groups[-1][-1] <= 2:
        groups[-1].append(y)
    else:
        groups.append([y])
axes = [int(np.mean(gp)) for gp in groups]
print("axis lines at y =", axes)
y_k = axes[1]  # middle axis: k in cm^2

def runs(mask_row, min_len):
    out, start = [], None
    for x, v in enumerate(mask_row):
        if v and start is None:
            start = x
        if not v and start is not None:
            if x - start >= min_len:
                out.append((start, x - 1))
            start = None
    if start is not None and len(mask_row) - start >= min_len:
        out.append((start, len(mask_row) - 1))
    return out

# Tick marks hang just below the axis line: short dark runs in a row a few px below.
below = dark[y_k + 5]
ticks = [(a + c) / 2 for a, c in runs(below, 1)]
print("k-axis ticks (px):", [round(t, 1) for t in ticks])
assert len(ticks) == 14, f"expected 14 decade ticks (1e-16 .. 1e-3 cm^2), found {len(ticks)}"
decades = np.arange(-16, -2)
slope, intercept = np.polyfit(decades, ticks, 1)
resid = np.array(ticks) - (slope * decades + intercept)
print(f"calibration: x = {intercept:.2f} + {slope:.3f} * log10(k_cm2); max tick residual {abs(resid).max():.2f} px")

def to_k(x):
    return 10 ** ((x - intercept) / slope)

BARS = {"Sandstone": 188, "Shale": 294, "Fractured Igneous & Metamorphic Rocks": 100, "Unweathered Marine Clay": 344}
for name, y0 in BARS.items():
    best = None
    for y in range(y0 - 14, y0 + 15):
        rr = runs(ink[y], 25)
        if len(rr) >= 1:
            width = rr[-1][1] - rr[0][0]
            if best is None or width > best[2]:
                best = (y, rr, width)
    y, rr, _ = best
    x0, x1 = rr[0][0], rr[-1][1]
    k0, k1 = to_k(x0), to_k(x1)
    print(f"{name}: row {y}, px {x0}..{x1}, k {k0:.2e}..{k1:.2e} cm^2 = {k0*1e-4:.2e}..{k1*1e-4:.2e} m^2, "
          f"log10 m^2 {np.log10(k0*1e-4):.2f}..{np.log10(k1*1e-4):.2f}")
